func DiagnoseOnIncident(d program.Diagnoser, dir string) (OwnIncidentResult, error)
    DiagnoseOnIncident runs d's Diagnose on the incident's captured open state:
    the controller's open@* snapshots in dir with their recorded reads, through
    the same RunDiagnose and read semantics the live stage uses. A learned
    playbook declares no object role, so it runs with no bindings; one that
    declares roles is unverified, because production binds them from findings
    the capture does not carry. A Match on any capture is a Match. Without one,
    a NoMatch on any capture is a failure; everything else is unverified.

type PrimitiveCall struct {
	// Phase separates observations made before and after repair. Empty means open.
	Phase     string          `json:"phase,omitempty"`
	Method    string          `json:"method"`
	Arguments json.RawMessage `json:"arguments"`
	Result    json.RawMessage `json:"result"`
	Error     string          `json:"error,omitempty"`
	// ErrorClass is the read class of Error (sdk.ReadErrorClass): absent,
	// refused, unavailable or invalid. A replay rebuilds an error of the same
	// class, so a program is judged on a recording exactly as it was live. A
	// recording with no class is legacy and reads as unavailable.
	ErrorClass string `json:"error_class,omitempty"`
	// FrameLimited marks a bounded host reply that takes precedence over snapshot fallbacks.
	FrameLimited bool `json:"frame_limited,omitempty"`
}
    PrimitiveCall stores one live ReadView call for deterministic replay.
